sequenceDiagram
    title Kostify - Alur Pembayaran Midtrans (Tenant)
    
    actor TenantUser as Tenant
    participant Dashboard as Tenant Dashboard
    participant ApiService as API Service
    participant Database as Database
    participant MidtransAPI as Midtrans API
    participant Browser as Browser

    rect rgb(200, 220, 255)
    note over TenantUser,Browser: User Initiates Payment
    Tenant->>Dashboard: Click "Bayar Sekarang"
    activate Dashboard
    
    Dashboard->>Database: Get pending payment
    activate Database
    Database-->>Dashboard: PaymentModel (pending status)
    deactivate Database
    
    Dashboard->>Dashboard: Validate payment
    Dashboard->>Dashboard: Generate Order ID<br/>(ORDER-{userId}-{timestamp})
    Dashboard->>Dashboard: Get customer name
    end
    
    rect rgb(200, 255, 220)
    note over Dashboard,MidtransAPI: Call Midtrans API
    Dashboard->>ApiService: getMidtransSnapUrl()<br/>(orderId, amount, customerName)
    activate ApiService
    
    ApiService->>ApiService: Check internet connection
    ApiService->>ApiService: Validate MIDTRANS_SERVER_KEY
    ApiService->>ApiService: Validate MIDTRANS_CLIENT_KEY
    ApiService->>ApiService: Encode Base64 credentials<br/>(ServerKey:)
    
    ApiService->>ApiService: Build payload:<br/>- transaction_details<br/>- customer_details<br/>- callbacks
    
    ApiService->>MidtransAPI: HTTP POST /v2/transactions<br/>(Basic Auth)
    activate MidtransAPI
    
    MidtransAPI->>MidtransAPI: Validate credentials
    MidtransAPI->>MidtransAPI: Create transaction token
    MidtransAPI->>MidtransAPI: Generate redirect_url
    
    MidtransAPI-->>ApiService: HTTP 201 Created<br/>{redirect_url, token_snap}
    deactivate MidtransAPI
    
    ApiService-->>Dashboard: MidtransResult.success(redirectUrl)
    deactivate ApiService
    end
    
    rect rgb(220, 255, 200)
    note over Dashboard,Database: Update Database
    Dashboard->>Database: updatePaymentRecord()<br/>(orderId, snapUrl, timestamp)
    activate Database
    Database-->>Dashboard: Payment updated
    deactivate Database
    end
    
    rect rgb(255, 220, 200)
    note over Dashboard,Browser: Open Payment Gateway
    Dashboard->>Browser: launchUrl(snapUrl)
    activate Browser
    Browser-->>Tenant: Midtrans Snap Payment Page
    end
    
    deactivate Dashboard
    
    rect rgb(255, 200, 200)
    note over Tenant,MidtransAPI: Process Payment
    Tenant->>Browser: Select payment method<br/>(Card/eWallet/Bank Transfer)
    Tenant->>Browser: Enter payment details
    
    Browser->>MidtransAPI: HTTP POST (Payment)
    activate MidtransAPI
    MidtransAPI->>MidtransAPI: Process payment<br/>- 3D Secure check<br/>- Fraud detection<br/>- Card validation
    MidtransAPI-->>Browser: HTTP 200<br/>{status: success/failed}
    deactivate MidtransAPI
    
    Browser-->>Tenant: Redirect to finish page
    deactivate Browser
    end
    
    rect rgb(200, 200, 255)
    note over Dashboard,MidtransAPI: Sync Payment Status (Later)
    Dashboard->>ApiService: _syncMidtransPayments()
    activate ApiService
    
    loop For each pending payment
        ApiService->>ApiService: Get payment.orderId
        ApiService->>MidtransAPI: HTTP GET /v2/{orderId}/status<br/>(Basic Auth)
        activate MidtransAPI
        MidtransAPI-->>ApiService: {transaction_status}
        deactivate MidtransAPI
        
        alt Status = "settlement" OR "capture"
            ApiService->>Database: updatePaymentStatus()<br/>(id, PaymentStatus.PAID)
            activate Database
            Database-->>ApiService: Updated
            deactivate Database
        end
    end
    
    ApiService-->>Dashboard: Status synced
    deactivate ApiService
    end
    
    Dashboard->>Tenant: Payment shown as "Lunas" ✓

sequenceDiagram
    title Kostify - Alur OCR KTP Scanner (Admin)
    
    actor AdminUser as Admin
    participant Screen as AddTenantScreen
    participant Camera as KtpCamera
    participant Picker as ImagePicker
    participant Compress as FlutterCompress
    participant MLKit as GoogleMLKit
    participant Database as Database

    rect rgb(200, 220, 255)
    note over AdminUser,Screen: Select Image Source
    Admin->>Screen: Click "Scan KTP"
    activate Screen
    Screen->>Admin: Show source picker dialog
    Admin->>Screen: Select "Camera"
    end
    
    rect rgb(200, 255, 220)
    note over Screen,Camera: Initialize Camera
    Screen->>Camera: Navigate to KtpCameraCaptureScreen
    activate Camera
    Camera->>Camera: initState()
    Camera->>Camera: availableCameras()
    Camera->>Camera: Select back camera
    Camera->>Camera: CameraController.initialize()
    Camera-->>Admin: Display camera preview<br/>(with KTP guide frame)
    end
    
    rect rgb(220, 255, 200)
    note over AdminUser,Camera: Capture Image
    Admin->>Camera: Click "Capture"
    Camera->>Camera: takePicture()
    Camera-->>Screen: Return image path
    deactivate Camera
    end
    
    rect rgb(255, 220, 200)
    note over Screen,Compress: Compress Image
    Screen->>Compress: FlutterImageCompress.compressWithFile()
    activate Compress
    Compress->>Compress: Auto-fix EXIF rotation
    Compress->>Compress: Compress to JPEG (quality: 80)
    Compress->>Compress: minWidth: 1024px
    Compress->>Compress: keepExif: false (strip metadata)
    Compress-->>Screen: Compressed Uint8List
    deactivate Compress
    
    Screen->>Screen: _cropToGuideBox()
    Screen->>Screen: Calculate KTP aspect ratio (1.58)
    Screen->>Screen: Crop: 88% width, 72% height max
    Screen->>Screen: Center crop for optimal recognition
    Screen-->>Screen: Cropped image ready
    end
    
    rect rgb(255, 200, 200)
    note over Screen,MLKit: OCR Processing
    Screen->>MLKit: Initialize TextRecognizer(script: LATIN)
    activate MLKit
    
    Screen->>MLKit: recognizer.processImage(inputImage)
    activate MLKit
    MLKit->>MLKit: Load ML Model
    MLKit->>MLKit: Text Detection<br/>(find text regions)
    MLKit->>MLKit: Character Recognition<br/>(recognize each char)
    MLKit->>MLKit: Calculate bounding boxes<br/>(text positions)
    MLKit-->>Screen: RecognizedText Object<br/>{blocks[], lines[], text}
    deactivate MLKit
    
    Screen->>MLKit: recognizer.close()
    deactivate MLKit
    end
    
    rect rgb(200, 200, 255)
    note over Screen,Screen: Parse OCR Result
    Screen->>Screen: _parseKTPResult(recognizedText)
    Screen->>Screen: Extract OCR lines<br/>(text + boundingBox)
    Screen->>Screen: Sort by vertical position<br/>(top to bottom)
    Screen->>Screen: Find field labels<br/>- _isNikLabel()?<br/>- _isNamaLabel()?<br/>- _isAlamatLabel()?
    
    Screen->>Screen: Extract NIK value<br/>_extractNikByLayout()<br/>Fallback: _extractNikFromRawText()
    Screen->>Screen: Extract Nama<br/>_extractSingleValueByAlignment()
    Screen->>Screen: Extract Alamat (multi-line)<br/>_extractAlamatComposite()
    
    Screen->>Screen: Create debug boxes<br/>(colored rectangles)
    Screen->>Screen: _setDebugImage()<br/>(for visualization)
    end
    
    rect rgb(200, 255, 200)
    note over Screen,Admin: Display & Verify
    Screen-->>Admin: Show extracted data:<br/>- NIK<br/>- Nama<br/>- Alamat
    Screen-->>Admin: Show debug visualization<br/>(colored boxes on image)
    
    Admin->>Screen: Verify OCR result
    
    alt Data is correct
        Admin->>Screen: Click "Lanjut" (Next)
        Screen->>Screen: Validate extracted data
        Screen-->>Admin: Move to next step
    else Data has errors
        Admin->>Screen: Manual edit text fields
        Admin->>Screen: Click "Lanjut"
        Screen-->>Admin: Continue with corrected data
    end
    end
    
    deactivate Screen

KOSTIFY - STATE DIAGRAM (Payment Status) (Mermaid Format)
================================================================================

stateDiagram-v2
    [*] --> PENDING: Admin creates payment

    PENDING --> PENDING: Tenant initiates payment<br/>(orderId, snapUrl updated)

    PENDING --> PAID: checkMidtransTransactionStatus()<br/>status = settlement OR capture

    PENDING --> PAID: Admin manually<br/>marks as paid

    PENDING --> OVERDUE: Current date > due date

    PAID --> [*]: Payment completed

    OVERDUE --> PAID: Payment made after due date

graph TD
    Start([Start: Captured Image]) --> Compress["FlutterImageCompress<br/>- Quality: 80%<br/>- Auto-fix EXIF<br/>- minWidth: 1024px"]

    Process --> ParseLines["Extract OCR Lines<br/>(text + boundingBox)"]
    
    ParseLines --> Sort["Sort by vertical<br/>position (top↓bottom)"]
    
    Sort --> FindLabels["Find Field Labels<br/>- NIK<br/>- NAMA LENGKAP<br/>- ALAMAT"]
    
    FindLabels --> ExtractNIK["_extractNikByLayout()<br/>- Search 16-digit number<br/>- Near NIK label"]
    
    ExtractNIK --> NIKCheck{Found?}
    NIKCheck -->|No| NIKFallback["_extractNikFromRawText()<br/>Regex: \\d{16}"]
    NIKCheck -->|Yes| ExtractNama
    NIKFallback --> ExtractNama
    
    ExtractNama["_extractSingleValueByAlignment()<br/>- Align right of label<br/>- Get name value"]
    
    ExtractNama --> ExtractAlamat["_extractAlamatComposite()<br/>- Multi-line collection<br/>- Join with spaces"]
    
    ExtractAlamat --> CreateDebug["Create debug boxes<br/>- White: All text<br/>- Blue: NIK<br/>- Purple: Nama<br/>- Orange: Alamat"]
    
    CreateDebug --> PopulateForm["Populate form fields<br/>automatically:<br/>- _nikCtrl<br/>- _namaCtrl<br/>- _alamatCtrl"]

    DisplayResult --> Verify{Admin verifies<br/>data OK?}
    
    Verify -->|Yes| Continue["Continue to next step"]
    Verify -->|No| ManualEdit["Admin manually edits"]
