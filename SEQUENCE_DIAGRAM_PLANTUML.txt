@startuml Payment_Midtrans_Flow
title Kostify - Alur Pembayaran Midtrans (Tenant)
actor TenantUser as tenant
participant TenantDashboard as dashboard
participant ApiService as api
database Database as db
participant "Midtrans API" as midtrans
actor "External Browser" as browser

== User Inisiasi Pembayaran ==
tenant -> dashboard: Click "Bayar Sekarang"
activate dashboard

dashboard -> db: Get pending payment
activate db
db --> dashboard: PaymentModel (pending)
deactivate db

dashboard -> dashboard: Validate payment exists
dashboard -> api: getMidtransSnapUrl()
activate api

== Validasi Kredensial ==
api -> api: Check internet connection
api -> api: Validate MIDTRANS_SERVER_KEY
api -> api: Validate MIDTRANS_CLIENT_KEY
api -> api: Encode Base64 credentials

== Buat Transaksi ==
api -> api: Generate Order ID\n(ORDER-{userId}-{timestamp})
api -> api: Build payload:\ntransaction_details,\ncustomer_details,\ncallbacks
api -> midtrans: POST /v2/transactions\n(Basic Auth)
activate midtrans

midtrans -> midtrans: Validate credentials
midtrans -> midtrans: Create transaction token
midtrans -> midtrans: Generate redirect_url
midtrans --> api: HTTP 201 Created\n{redirect_url, token_snap}
deactivate midtrans

api --> dashboard: MidtransResult.success(redirectUrl)
deactivate api

== Update Database ==
dashboard -> db: updatePaymentRecord()\n(orderId, snapUrl)
activate db
db --> dashboard: Updated
deactivate db

== Buka Payment Gateway ==
dashboard -> browser: launchUrl(snapUrl)
activate browser
browser --> tenant: Midtrans Snap Payment Page
deactivate browser
deactivate dashboard

== Proses Pembayaran ==
tenant -> browser: Fill payment details\n(Card/eWallet/Bank Transfer)
browser -> midtrans: HTTP POST Payment
activate midtrans
midtrans -> midtrans: Process payment\n(3D Secure, Fraud Check)
midtrans --> browser: HTTP 200\n{status: success/failed}
deactivate midtrans

browser --> tenant: Redirect to finish page
deactivate browser

== Sync Status (Later) ==
dashboard -> api: _syncMidtransPayments()
activate api

loop For each pending payment
  api -> api: Get payment.orderId
  api -> midtrans: GET /v2/{orderId}/status
  activate midtrans
  midtrans --> api: {transaction_status}
  deactivate midtrans
  
  alt Status == settlement OR capture
    api -> db: updatePaymentStatus()\n(PaymentStatus.PAID)
    activate db
    db --> api: Updated
    deactivate db
  end
end

api --> dashboard: Status synced
deactivate api

dashboard -> tenant: Payment shown as "Lunas"

@enduml

@startuml OCR_KTP_Flow
title Kostify - Alur OCR KTP Scanner (Admin)
actor AdminUser as admin
participant AddTenantScreen as screen
participant KtpCamera as camera
participant ImagePicker as picker
participant FlutterCompress as compress
participant GoogleMLKit as mlkit
database Database as db

== Pilih Sumber ==
admin -> screen: Click "Scan KTP"
activate screen
screen -> admin: Show source picker dialog
admin -> screen: Select "Camera"

== Inisialisasi Camera ==
screen -> camera: Navigate to KtpCameraCaptureScreen
activate camera
camera -> camera: initState()
camera -> camera: availableCameras()
camera -> camera: Select back camera
camera -> camera: Initialize CameraController
camera --> admin: Display camera preview\n(with KTP guide frame)

== Capture Image ==
admin -> camera: Click "Capture"
camera -> camera: takePicture()
camera --> screen: Return image path
deactivate camera

== Compress & Crop ==
screen -> compress: FlutterImageCompress.compressWithFile()
activate compress
compress -> compress: Auto-fix EXIF rotation
compress -> compress: Compress to JPEG (quality: 80)
compress -> compress: minWidth: 1024, keepExif: false
compress --> screen: Compressed Uint8List
deactivate compress

screen -> screen: _cropToGuideBox()
screen -> screen: Calculate KTP aspect ratio (1.58)
screen -> screen: Crop 88% width, 72% height
screen --> screen: Cropped image

== OCR Processing ==
screen -> mlkit: Initialize TextRecognizer\n(script: LATIN)
activate mlkit
screen -> mlkit: recognizer.processImage()
activate mlkit

mlkit -> mlkit: Load ML Model
mlkit -> mlkit: Text Detection (find regions)
mlkit -> mlkit: Character Recognition
mlkit -> mlkit: Calculate bounding boxes
mlkit --> screen: RecognizedText Object
deactivate mlkit
deactivate mlkit

== Parse Result ==
screen -> screen: _parseKTPResult(recognizedText)
screen -> screen: Extract OCR lines\n(text + boundingBox)
screen -> screen: Sort by vertical position
screen -> screen: Find field labels\n(NIK, NAMA, ALAMAT)
screen -> screen: Extract values by alignment

alt NIK extraction
  screen -> screen: _extractNikByLayout()
  screen -> screen: Fallback: _extractNikFromRawText()
end

screen -> screen: _extractSingleValueByAlignment()\n(for Nama)
screen -> screen: _extractAlamatComposite()\n(multi-line for Alamat)

== Display Result ==
screen -> screen: Create debug boxes\n(colored rectangles for visualization)
screen -> screen: Populate form fields\nautomatically
screen --> admin: Show extracted data\n(with debug visualization)

== Verify & Continue ==
admin -> screen: Verify OCR result
alt Data correct
  admin -> screen: Click "Lanjut" (Next)
  screen -> screen: Validate extracted data
  screen --> admin: Move to next step
else Data incorrect
  admin -> screen: Manual edit text fields
  admin -> screen: Click "Lanjut"
end

deactivate screen

@enduml
